"""Bounded FIFO/exec observation proof, not mapped credential or engine qualification."""
from pathlib import Path
import hashlib,json,os,secrets,select,shutil,subprocess,tempfile,time

OUT=Path(__file__).resolve().parent
WORK=Path(tempfile.mkdtemp(prefix='reploy-fg-exec-status-'))
RESULT={'scope':'Linux private status FIFO + raw nonleader exec/CLOEXEC mechanism only; no production tail change, credential/capability qualification, engine transport or public startup claim','cases':[],'mechanism':'existing close_range/UNSHARE + raw dup3(O_CLOEXEC) + immediate close_range4; raw nonblocking descriptor ownership; fd3 and higher-source layouts'}
start=time.monotonic()
children=[]
def exact(fd,n,deadline):
 result=b''
 while len(result)<n:
  if time.monotonic()>deadline:raise TimeoutError('bounded read')
  select.select([fd],[],[],min(.1,max(0,deadline-time.monotonic())))
  try:chunk=os.read(fd,n-len(result))
  except BlockingIOError:continue
  if chunk:result+=chunk
 return result
try:
 binary=WORK/'probe'
 subprocess.run(['go','build','-o',str(binary),str(OUT/'startup_status_reservedfd_probe.go')],check=True,timeout=90,capture_output=True)
 assembly=subprocess.run(['go','tool','objdump','-s','main.execTail',str(binary)],check=True,timeout=20,capture_output=True,text=True).stdout
 calls=[line.strip() for line in assembly.splitlines() if 'CALL' in line]
 assert 'TEXT ' in assembly and calls and all('syscall.RawSyscall' in line for line in calls) and 'runtime.' not in assembly
 (OUT/'startup-status-reserved-fd-tail.txt').write_text(assembly)
 RESULT['compiler']=subprocess.run(['go','version'],check=True,capture_output=True,text=True).stdout.strip()
 RESULT['binary_sha256']=hashlib.sha256(binary.read_bytes()).hexdigest()
 RESULT['compiled_tail']={'raw_call_count':len(calls),'no_runtime_calls':True,'disassembly_sha256':hashlib.sha256(assembly.encode()).hexdigest()}
 for mode in ['success','success-low-fd','exec-failure','before-exec-failure','cancel-before-release']:
  fifo=WORK/(mode+'.fifo');os.mkfifo(fifo,0o600)
  canary=WORK/(mode+'.private');canary.write_bytes(b'not-an-application-descriptor')
  token=secrets.token_hex(8);fd=os.open(fifo,os.O_RDONLY|os.O_NONBLOCK)
  try:
   p=subprocess.Popen([str(binary),mode,str(fifo),token,str(canary),'fixture'],stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
   children.append(p);deadline=time.monotonic()+10
   try:header=exact(fd,24,deadline)
   except Exception:
    if p.poll() is not None:RESULT['early_child_failure']={'mode':mode,'exit_code':p.returncode,'stderr':p.stderr.read().decode(),'stdout':p.stdout.read().decode()}
    raise
   assert header==b'RPLYEX01'+token.encode()
   assert select.select([p.stdout],[],[],5)[0] and p.stdout.readline()==b'ARMED\n'
   fifo.unlink();assert not fifo.exists()
   # A live inhibited process is not startup completion. Its writer is held.
   try:
    premature=os.read(fd,16)
    raise AssertionError('unexpected status data/EOF before release: '+repr(premature))
   except BlockingIOError:pass
   if mode=='cancel-before-release':p.kill();p.wait(timeout=5)
   else:p.stdin.write(b'G');p.stdin.flush()
   body=b'';eof=False
   while time.monotonic()<deadline:
    select.select([fd],[],[],.05)
    try:chunk=os.read(fd,16)
    except BlockingIOError:continue
    if not chunk:eof=True;break
    body+=chunk;assert len(body)<=4
   assert eof
   live=p.poll() is None
   # EOF alone is ambiguous with death. This is only a fixture observation:
   # production must use its held proc-directory/pidfd and exact start binding.
   try:
    image_fd=os.open(f'/proc/{p.pid}/exe',os.O_PATH|os.O_CLOEXEC)
    kernel_image=bool(os.fstat(image_fd).st_ino)
    os.close(image_fd)
   except (FileNotFoundError,ProcessLookupError):kernel_image=False
   observation={'mode':mode,'bound_header':True,'unlinked_before_release':True,'inhibited_no_ack':True,'status_eof':eof,'errno_bytes':body.hex(),'live_after_eof':live,'kernel_image_after_eof':kernel_image,'accepted_start':not body and live and kernel_image}
   if mode in ['success','success-low-fd']:
    assert observation['accepted_start']
    assert select.select([p.stdout],[],[],5)[0]
    app=json.loads(p.stdout.readline());assert app['status_path_absent'] and app['private_descriptor_leaks']==[]
    observation['application_control']=app
    p.stdin.write(b'Q');p.stdin.flush()
   else:
    p.wait(timeout=5)
    observation['accepted_start']=not body and p.poll() is None and kernel_image
    assert not observation['accepted_start']
    if mode=='exec-failure':assert len(body)==4
   p.wait(timeout=5);observation['exit_code']=p.returncode
   observation['stderr']=p.stderr.read().decode()
   RESULT['cases'].append(observation)
  finally:os.close(fd)
 RESULT['passed']=len(RESULT['cases'])==5
except Exception as e:RESULT['error']=repr(e)
finally:
 for p in children:
  if p.poll() is None:p.kill();p.wait(timeout=5)
 shutil.rmtree(WORK)
 RESULT['private_root_removed']=not WORK.exists()
 RESULT['elapsed_seconds']=time.monotonic()-start
 RESULT['source_sha256']={p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in [Path(__file__),OUT/'startup_status_reservedfd_probe.go']}
 (OUT/'startup-status-reserved-fd-proof-result.json').write_text(json.dumps(RESULT,indent=2)+'\n')
 print(json.dumps(RESULT,indent=2))
raise SystemExit(0 if RESULT.get('passed') else 1)
